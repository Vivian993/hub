    <script>
        // --- 1. 時間與日期處理 ---
        const days = ['星期日', '星期一', '星期二', '星期三', '星期四', '星期五', '星期六'];
        
        function updateTime() {
            const now = new Date();
            const dateString = `${now.getMonth() + 1}月${now.getDate()}日｜${days[now.getDay()]}`;
            document.getElementById('current-date').innerText = dateString;
            return now;
        }

        // 動態推算未來的假發車時間與抵達時間
        function generateDynamicTimes() {
            const now = new Date();
            const times = [];
            let offset = Math.floor(Math.random() * 5) + 3;

            for(let i=0; i<3; i++) {
                const departTime = new Date(now.getTime() + offset * 60000);
                const arriveTime = new Date(departTime.getTime() + 18 * 60000);

                const depHrs = departTime.getHours().toString().padStart(2, '0');
                const depMins = departTime.getMinutes().toString().padStart(2, '0');
                const arrHrs = arriveTime.getHours().toString().padStart(2, '0');
                const arrMins = arriveTime.getMinutes().toString().padStart(2, '0');

                times.push({
                    timeStr: `${depHrs}:${depMins}`,
                    arriveStr: `${arrHrs}:${arrMins}`,
                    minutesLeft: offset
                });
                offset += Math.floor(Math.random() * 8) + 8;
            }
            return times;
        }

        // --- 2. 桃園真實站牌擴充資料庫 ---
        const taoyuanStopsDB = [
            { id: '1', name: '愛買(中山路)', lat: 24.9865, lng: 121.2952, mainRoute: '1路', allRoutes: '1路, 139路, 307路' },
            { id: '2', name: '桃園農工', lat: 24.9959, lng: 121.3195, mainRoute: '1路', allRoutes: '1路, 137路, 302路' },
            { id: '3', name: '武陵高中', lat: 24.9894, lng: 121.2965, mainRoute: '1路', allRoutes: '1路, 139路, 601路' },
            { id: '4', name: '中平路口', lat: 24.9902, lng: 121.3021, mainRoute: '1路', allRoutes: '1路, 220路' },
            { id: '5', name: '中山國小', lat: 24.9875, lng: 121.2901, mainRoute: '307路', allRoutes: '1路, 307路' },
            { id: '6', name: '桃園火車站', lat: 24.9892, lng: 121.3135, mainRoute: '1路', allRoutes: '各路幹線公車' },
            { id: '7', name: '龍安街口', lat: 24.9840, lng: 121.2890, mainRoute: '307路', allRoutes: '307路, 1B路' }
        ];

        // 哈維辛公式計算距離
        function calculateDistance(lat1, lon1, lat2, lon2) {
            const R = 6371e3;
            const φ1 = lat1 * Math.PI/180;
            const φ2 = lat2 * Math.PI/180;
            const Δφ = (lat2-lat1) * Math.PI/180;
            const Δλ = (lon2-lon1) * Math.PI/180;
            const a = Math.sin(Δφ/2) * Math.sin(Δφ/2) +
                      Math.cos(φ1) * Math.cos(φ2) *
                      Math.sin(Δλ/2) * Math.sin(Δλ/2);
            const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
            return Math.round(R * c);
        }

        // 全域變數
        let processedStops = [];
        let currentSelectedStop = null;
        let currentMode = 'home'; // 'home' 或 'school'

        function requestLocation() {
            document.getElementById('location-text').innerText = "定位中...";
            document.getElementById('gps-icon').classList.add('animate-spin');

            if (navigator.geolocation) {
                navigator.geolocation.getCurrentPosition(
                    position => {
                        document.getElementById('gps-icon').classList.remove('animate-spin');
                        const userLat = position.coords.latitude;
                        const userLng = position.coords.longitude;
                        
                        const isTaoyuan = (userLat > 24.8 && userLat < 25.1 && userLng > 121.1 && userLng < 121.4);
                        const finalLat = isTaoyuan ? userLat : 24.9880; 
                        const finalLng = isTaoyuan ? userLng : 121.2960;

                        document.getElementById('location-text').innerText = isTaoyuan ? "已精準定位" : "模擬桃園市區定位";
                        processStops(finalLat, finalLng);
                    },
                    error => {
                        document.getElementById('gps-icon').classList.remove('animate-spin');
                        document.getElementById('location-text').innerText = "預設定位 (愛買附近)";
                        processStops(24.9865, 121.2952);
                    },
                    { enableHighAccuracy: true, timeout: 5000 }
                );
            }
        }

        function processStops(lat, lng) {
            processedStops = taoyuanStopsDB.map(stop => {
                const dist = calculateDistance(lat, lng, stop.lat, stop.lng);
                const walkMins = Math.max(1, Math.round(dist / 60));
                const dynamicTimes = generateDynamicTimes();
                return { ...stop, distance: dist, walkMins: walkMins, times: dynamicTimes };
            });

            processedStops.sort((a, b) => a.distance - b.distance);
            
            // 預設選擇第一個最近的站牌
            if (!currentSelectedStop) {
                currentSelectedStop = processedStops[0];
            }

            renderNearbyList(processedStops);
            updateRecommendCard(currentSelectedStop);
        }

        // --- 🔍 搜尋功能邏輯 ---
        function handleSearch() {
            const query = document.getElementById('search-input').value.trim().toLowerCase();
            const clearBtn = document.getElementById('clear-search-btn');

            if (query.length > 0) {
                clearBtn.classList.remove('hidden');
                document.getElementById('list-title').innerHTML = `<i class="fa-solid fa-magnifying-glass text-amber-500"></i> 搜尋結果 (點擊選擇上車)`;
                
                const filtered = processedStops.filter(stop => 
                    stop.name.toLowerCase().includes(query) || 
                    stop.allRoutes.toLowerCase().includes(query)
                );
                renderNearbyList(filtered);
            } else {
                clearSearch();
            }
        }

        function clearSearch() {
            document.getElementById('search-input').value = '';
            document.getElementById('clear-search-btn').classList.add('hidden');
            document.getElementById('list-title').innerHTML = `<i class="fa-solid fa-location-arrow text-blue-500"></i> 您附近的站牌 (點擊選擇上車)`;
            renderNearbyList(processedStops);
        }

        function focusSearch() {
            const searchInput = document.getElementById('search-input');
            searchInput.focus();
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // 渲染站牌清單，並加上「點擊選擇上車」事件
        function renderNearbyList(stops) {
            const container = document.getElementById('nearby-stops-list');

            if (stops.length === 0) {
                container.innerHTML = `
                    <div class="text-center p-6 text-slate-400 text-sm">
                        <i class="fa-solid fa-face-frown text-3xl mb-2 text-slate-300"></i><br>
                        找不到相關的站牌或公車號碼喔！
                    </div>
                `;
                return;
            }

            container.innerHTML = stops.map(stop => {
                const isSelected = currentSelectedStop && currentSelectedStop.id === stop.id;
                return `
                    <div onclick="selectBoardingStop('${stop.id}')" class="flex items-center justify-between p-3.5 rounded-xl border transition cursor-pointer active:scale-95 ${isSelected ? 'bg-emerald-100/80 border-emerald-500 ring-2 ring-emerald-400' : 'bg-slate-50 hover:bg-emerald-50 border-slate-100'}">
                        <div class="flex items-center gap-3">
                            <div class="${isSelected ? 'bg-emerald-600 text-white' : 'bg-blue-100 text-blue-700'} p-2 rounded-lg font-bold text-xs flex flex-col items-center w-14">
                                <span class="text-[10px] ${isSelected ? 'text-emerald-100' : 'text-blue-500'}">${isSelected ? '選取中' : '距離'}</span>
                                ${stop.distance > 999 ? (stop.distance/1000).toFixed(1)+'km' : stop.distance+'m'}
                            </div>
                            <div>
                                <div class="flex items-center gap-2">
                                    <h4 class="font-bold text-slate-800 text-base">${stop.name}</h4>
                                    <span class="bg-amber-100 text-amber-800 text-[10px] font-black px-1.5 py-0.5 rounded">${stop.mainRoute}</span>
                                </div>
                                <span class="text-xs text-slate-500">🚶 走 ${stop.walkMins} 分鐘 | 經過: ${stop.allRoutes}</span>
                            </div>
                        </div>
                        <div class="text-right">
                            <span class="text-lg font-black text-emerald-600">${stop.times[0].timeStr}</span>
                            <span class="text-[10px] block text-emerald-500 font-bold">${stop.times[0].minutesLeft} 分後進站</span>
                        </div>
                    </div>
                `;
            }).join('');
        }

        // 點擊卡片選擇上車站牌
        function selectBoardingStop(stopId) {
            const found = processedStops.find(s => s.id === stopId);
            if (found) {
                currentSelectedStop = found;
                updateRecommendCard(currentSelectedStop);
                // 重新渲染清單標示選取狀態
                const query = document.getElementById('search-input').value.trim();
                if (query.length > 0) {
                    handleSearch();
                } else {
                    renderNearbyList(processedStops);
                }
            }
        }

        function updateRecommendCard(stop) {
            if(!stop) return;
            const nextBus = stop.times[0];

            if (currentMode === 'school') {
                document.getElementById('recommend-route').innerText = stop.mainRoute === "307路" ? "307路" : "1路 / 302路";
                document.getElementById('recommend-board').innerText = stop.name;
                document.getElementById('recommend-alight').innerText = "桃園農工";
                document.getElementById('recommend-direction').innerText = "往 桃園農工/萬壽路 方向";
            } else {
                document.getElementById('recommend-route').innerText = stop.mainRoute === "307路" ? "307路" : "1路 / 307路";
                document.getElementById('recommend-board').innerText = stop.name;
                document.getElementById('recommend-alight').innerText = "愛買(中山路)";
                document.getElementById('recommend-direction').innerText = "往 中壢/中山路 方向";
            }

            document.getElementById('recommend-time').innerText = nextBus.timeStr;
            document.getElementById('recommend-board-time').innerText = nextBus.timeStr;
            document.getElementById('recommend-arrive-time').innerText = `預計 ${nextBus.arriveStr} 到達`;
            document.getElementById('recommend-walk').innerHTML = `<i class="fa-solid fa-person-walking"></i> 步行約 ${stop.walkMins} 分鐘`;
        }

        // --- 4. 模式切換 ---
        function startMode(type) {
            currentMode = type;
            if(!currentSelectedStop) return alert('正在搜尋站牌中，請稍等喔！');

            updateRecommendCard(currentSelectedStop);

            if (type === 'school') {
                showModal('🏫 去學校模式', document.getElementById('recommend-route').innerText, currentSelectedStop.name, '桃園農工', currentSelectedStop.times[0].timeStr, currentSelectedStop.times[0].arriveStr, '往 桃園農工/萬壽路 方向');
            } else {
                showModal('🏠 回家模式', document.getElementById('recommend-route').innerText, currentSelectedStop.name, '愛買(中山路)', currentSelectedStop.times[0].timeStr, currentSelectedStop.times[0].arriveStr, '往 中壢/中山路 方向');
            }
        }

        function confirmTrip() {
            if(!currentSelectedStop) return;
            const route = document.getElementById('recommend-route').innerText;
            const board = document.getElementById('recommend-board').innerText;
            const alight = document.getElementById('recommend-alight').innerText;
            const time = document.getElementById('recommend-time').innerText;
            const arriveTime = document.getElementById('recommend-arrive-time').innerText;
            const direction = document.getElementById('recommend-direction').innerText;
